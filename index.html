/** * Logistics Dashboard — Google Sheet database (Apps Script Web App) * * วิธีติดตั้ง (ทำครั้งเดียว โดยเจ้าของ Google Sheet): * 1. เปิด Google Sheet "Logistics Supply Chain" → เมนู ส่วนขยาย (Extensions) → Apps Script * 2. ลบโค้ดเดิมในไฟล์ Code.gs แล้ววางโค้ดทั้งหมดนี้แทน * 3. แก้ UPLOAD_KEY ด้านล่างเป็นรหัสที่ต้องการ (ใช้ตอนอัปโหลด/ลบข้อมูลจาก Dashboard) * 4. กดบันทึก → กด "ทำให้ใช้งานได้" (Deploy) → "การทำให้ใช้งานได้รายการใหม่" (New deployment) * - ประเภท: เว็บแอป (Web app) * - เรียกใช้ในฐานะ (Execute as): ฉัน (Me) * - ผู้ที่มีสิทธิ์เข้าถึง (Who has access): ทุกคน (Anyone) * กด Deploy → อนุญาตสิทธิ์ (Authorize) → คัดลอก "URL ของเว็บแอป" ที่ลงท้ายด้วย /exec * 5. นำ URL ไปใส่ใน Dashboard (หน้าต่าง "เพิ่มข้อมูล" → ตั้งค่าการเชื่อมต่อ) หรือส่งให้ผู้ดูแลฝังในไฟล์ HTML * * หมายเหตุ: ถ้าแก้โค้ดนี้ภายหลัง ต้อง Deploy → Manage deployments → แก้ไข (ไอคอนดินสอ) → Version: New version * URL เดิมจะยังใช้ได้ * * การเก็บข้อมูล: Sheet "Dashboard_Data" (สร้างให้อัตโนมัติ) 1 แถว = 1 เดือน * ข้อมูลทั้งเดือนเก็บเป็น JSON แบ่งเป็นช่อง ๆ (ช่องละไม่เกิน 45,000 ตัวอักษร) — ห้ามแก้ไขช่อง JSON ด้วยมือ */ const UPLOAD_KEY = 'CHANGE-ME'; // ← เปลี่ยนเป็นรหัสอัปโหลดของคุณ const SHEET_NAME = 'Dashboard_Data'; const CHUNK = 45000; // Google Sheet จำกัด 50,000 ตัวอักษรต่อช่อง (ใช้ 45,000 + ตัวนำ '~') const FIXED = 7; // จำนวนคอลัมน์ก่อนช่อง JSON const HEAD = ['month', 'file', 'uploadedAt', 'coilStock (ตัน)', 'coilWarehouse (ตัน)', 'fgStock (ตัน)', 'chunks', 'json →']; function sheet_() { const ss = SpreadsheetApp.getActive(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.getRange(1, 1, 1, HEAD.length).setValues([HEAD]).setFontWeight('bold').setBackground('#0c3843').setFontColor('#ffffff'); sh.setFrozenRows(1); sh.getRange('A:A').setNumberFormat('@'); } return sh; } function out_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function rows_() { const sh = sheet_(); const n = sh.getLastRow() - 1; if (n < 1) return []; const w = Math.max(sh.getLastColumn(), HEAD.length); return sh.getRange(2, 1, n, w).getDisplayValues().map((r, i) => ({ row: i + 2, r: r })); } function find_(month) { return rows_().find(x => x.r[0] === month) || null; } function list_() { return rows_().filter(x => /^\d{4}-\d{2}$/.test(x.r[0])) .map(x => ({ month: x.r[0], file: x.r[1], uploadedAt: x.r[2] })) .sort((a, b) => a.month < b.month ? 1 : -1); } function read_(month) { const x = find_(month); if (!x) throw new Error('ไม่พบข้อมูลเดือน ' + month); const n = Number(x.r[6]) || 1; const json = x.r.slice(FIXED, FIXED + n).map(s => s.replace(/^~/, '')).join(''); return { month: month, file: x.r[1], uploadedAt: x.r[2], data: JSON.parse(json) }; } function doGet(e) { try { const p = (e && e.parameter) || {}; if (p.action === 'get') return out_(Object.assign({ ok: true }, read_(String(p.month)))); return out_({ ok: true, months: list_() }); } catch (err) { return out_({ ok: false, error: String(err && err.message || err) }); } } function doPost(e) { const lock = LockService.getScriptLock(); try { const body = JSON.parse((e && e.postData && e.postData.contents) || '{}'); if (UPLOAD_KEY === 'CHANGE-ME') throw new Error('ยังไม่ได้ตั้งรหัสอัปโหลดใน Apps Script (UPLOAD_KEY)'); if (body.key !== UPLOAD_KEY) throw new Error('รหัสอัปโหลดไม่ถูกต้อง'); const month = String(body.month || ''); if (!/^\d{4}-\d{2}$/.test(month)) throw new Error('รูปแบบเดือนไม่ถูกต้อง'); lock.waitLock(20000); const sh = sheet_(); if (body.action === 'delete') { const x = find_(month); if (x) sh.deleteRow(x.row); return out_({ ok: true, months: list_() }); } if (body.action === 'save') { const d = body.data; if (!d || !Array.isArray(d.coil) || !Array.isArray(d.fg)) throw new Error('ข้อมูลไม่ครบ'); const json = JSON.stringify(d); const chunks = []; // '~' prefix keeps every piece plain text (a piece starting with = + - would otherwise become a formula/number) for (let i = 0; i < json.length; i += CHUNK) chunks.push('~' + json.slice(i, i + CHUNK)); const sum = (arr, k) => arr.reduce((s, r) => s + (Number(r[k]) || 0), 0); const coilStock = sum(d.coil.filter(r => r.seg === 'All'), 'stock'); const wh = d.storageCoil && d.storageCoil[0] ? Number(d.storageCoil[0].stock) || 0 : 0; const values = [month, String(body.file || ''), new Date(), Math.round(coilStock), Math.round(wh), Math.round(sum(d.fg, 'stock')), chunks.length].concat(chunks); const found = find_(month); const row = found ? found.row : sh.getLastRow() + 1; const lastCol = Math.max(sh.getLastColumn(), values.length); sh.getRange(row, 1, 1, lastCol).clearContent(); sh.getRange(row, 1).setNumberFormat('@'); sh.getRange(row, 1, 1, values.length).setValues([values]); return out_({ ok: true, months: list_() }); } throw new Error('ไม่รู้จักคำสั่ง ' + body.action); } catch (err) { return out_({ ok: false, error: String(err && err.message || err) }); } finally { try { lock.releaseLock(); } catch (e2) {} } }
